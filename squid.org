#+title: RMI squid plot

In this notebook we study Alex Engel's squid plot, which examines the
electricity planning peak demand forecasting errors made by FERC714 respondents.

* Load in data

#+begin_src python :results none
from pathlib import Path

import matplotlib.pyplot as plt
import polars as pl
import seaborn as sns

sns.set_theme()

data_path = Path("data/")
respondents = pl.read_parquet(data_path / "core_ferc714__respondent_id.parquet")
hourly_demand = pl.read_parquet(
    data_path / "out_ferc714__hourly_planning_area_demand.parquet"
)
forecast = pl.read_parquet(
    data_path / "core_ferc714__yearly_planning_area_demand_forecast.parquet"

first_year = 2006

* Grow the squid

To create the squid plot we need to compare the forecasted peak demand to the
realized peak demand.
#+begin_src python
squid_all_years = named_forecast.join(
    peak_demand.select("respondent_id_ferc714", "year", "actual"),
    on=["respondent_id_ferc714", "year"],
    how="inner",
    validate="m:1",
)
squid_all_years = squid_all_years.with_columns(
    error=pl.when((pl.col("actual") != 0) & (pl.col("forecast") != 0))
    .then((pl.col("forecast") - pl.col("actual")) / pl.col("actual"))
    .otherwise(None)
).pivot(
    "lookahead",
    index=["respondent_id_ferc714", "respondent_name_ferc714", "year"],
    values="error",
)

# build an expression that checks whether all lookaheads are null
lookaheads_expr = [pl.col(str(i)) for i in range(11)]
all_nulls_expr = pl.lit(True)
for expr in lookaheads_expr:
    all_nulls_expr &= expr.is_null()
# drop any rows where there's no year-1 forecast
# and fill the year-0 forecast with 0
squid_all_years = squid_all_years.filter(
    ~all_nulls_expr, pl.col("1").is_not_null()
).with_columns(pl.lit(0).alias("0"))
print(squid_all_years)
#+end_src

forecast errors?

squid = squid_all_years.group_by(
